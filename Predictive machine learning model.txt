import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestRegressor
import matplotlib.pyplot as plt
from datetime import datetime, timedelta

# Step 1: Load the dataset
orders_df = pd.read_csv('predicted file.csv')

# Step 2: Preprocess the data
orders_df['order_purchase_timestamp'] = pd.to_datetime(orders_df['order_purchase_timestamp'])
orders_df['order_date'] = orders_df['order_purchase_timestamp'].dt.date
sales_data = orders_df.groupby(['order_date', 'product_category_name'])['payment_value'].sum().reset_index()

# Include order_date as a feature
sales_data['order_date'] = pd.to_datetime(sales_data['order_date'])
sales_data['year'] = sales_data['order_date'].dt.year
sales_data['month'] = sales_data['order_date'].dt.month
sales_data['day'] = sales_data['order_date'].dt.day

# Step 3: Train a machine learning model
X = pd.get_dummies(sales_data[['product_category_name', 'year', 'month', 'day']])
y = sales_data['payment_value']
model = RandomForestRegressor()
model.fit(X, y)

# Define a function to generate dates for the next 6 months
def generate_future_dates(start_date, num_months):
    future_dates = [start_date + timedelta(days=30*i) for i in range(1, num_months+1)]
    return future_dates

# Define the start date as the last date in your dataset
start_date = sales_data['order_date'].max()

# Generate future dates for the next 6 months
future_dates = generate_future_dates(start_date, 6)

# Initialize a list to hold the expanded dates
expanded_dates = []

# Iterate over unique product categories
for category in sales_data['product_category_name'].unique():
    # Repeat each date for the number of unique categories
    expanded_dates.extend(future_dates)

# Repeat each category for the total number of dates
expanded_categories = list(sales_data['product_category_name'].unique()) * len(future_dates)

# Create a DataFrame with expanded dates and product categories
future_data = pd.DataFrame({
    'order_date': expanded_dates,
    'product_category_name': expanded_categories
})

# Extract year, month, and day from order_date
future_data['year'] = future_data['order_date'].dt.year
future_data['month'] = future_data['order_date'].dt.month
future_data['day'] = future_data['order_date'].dt.day

# One-hot encode product categories
future_data = pd.get_dummies(future_data[['product_category_name', 'year', 'month', 'day']])

# Make predictions
predictions = model.predict(future_data)

# Combine predictions with future dates
predictions_df = pd.DataFrame({
    'order_date': expanded_dates,
    'product_category_name': expanded_categories,
    'predicted_payment_value': predictions
})

# Print or visualize predictions
print(predictions_df)

# Print or visualize predictions
predictions_df.to_csv("PreditionSales.csv")
